
## 训练命令
```bash
set -e

PROJECT_ROOT=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/RL-NavGPT
TRAIN_PYTHON=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/venvs/navgpt-rl/bin/python
GPU_LIST="0,1"

cd "$PROJECT_ROOT/nav_src"

LEVEL2_REPORT="$PROJECT_ROOT/outputs/stage6-four-level-20260808-063623/level2_report.json"

test -f "$LEVEL2_REPORT"

AUDITED_MAX_COMPLETION_LENGTH=$(
  "$TRAIN_PYTHON" -c \
  'import json,sys; print(json.load(open(sys.argv[1]))["token_budget"]["recommended_max_completion_length"])' \
  "$LEVEL2_REPORT"
)

TRAIN_MAX_COMPLETION_LENGTH=8192

test "$TRAIN_MAX_COMPLETION_LENGTH" -le \
  "$AUDITED_MAX_COMPLETION_LENGTH"

RUN_TAG="main8.1-ours-qwen14b-fullrl-seed0-2gpu-750steps-lr2e6-$(date +%Y%m%d-%H%M%S)"

OUTPUT_DIR="$PROJECT_ROOT/outputs/$RUN_TAG"
CONSOLE_LOG="${OUTPUT_DIR}.console.log"

test ! -e "$OUTPUT_DIR"
test ! -e "$CONSOLE_LOG"

mkdir -p "$PROJECT_ROOT/outputs"

# 保存实验代码身份
git -C "$PROJECT_ROOT" rev-parse HEAD \
  > "${OUTPUT_DIR}.git-commit.txt"

git -C "$PROJECT_ROOT" status --short \
  > "${OUTPUT_DIR}.git-status.txt"

git -C "$PROJECT_ROOT" diff \
  > "${OUTPUT_DIR}.code.patch"

TRAIN_ARGS=(
  scripts/launch_grpo.py

  --mode ddp
  --gpus "$GPU_LIST"
  --nccl-profile default

  --

  --output-dir "$OUTPUT_DIR"

  --process-group-timeout-seconds 7200

  --expected-instruction-count 14039

  --clip-text-device auto
  --clip-text-dtype fp16

  --max-navigation-steps 10

  --r 16
  --lora-alpha 32
  --lora-dropout 0.05
  --max-trainable-percentage 1.0

  --dtype bf16

  --max-completion-length "$TRAIN_MAX_COMPLETION_LENGTH"

  --num-generations 4

  --learning-rate 2e-6
  --weight-decay 0.0
  --warmup-ratio 0.03
  --max-grad-norm 1.0
  --lr-scheduler-type cosine

  --beta 0.001

  --temperature 1.0
  --top-p 0.95

  --max-tool-calling-iterations 10

  --trainer-max-steps 750
  --num-train-epochs 1.0

  --logging-steps 5

  --save-steps 50
  --save-total-limit 3

  --trajectory-log-interval 50

  --validation

  --validation-annotation \
    ../datasets/R2R/annotations/R2R_val_unseen_instr.json

  --validation-action-plan-cache \
    ../datasets/R2R/action_plan_cache/qwen2.5-14b-val-unseen-t0-v1/R2R_val_unseen_action_plans.jsonl

  --validation-fast-subset-manifest \
    ../datasets/R2R/eval_subsets/R2R_val_unseen_fast128_seed0.json

  --validation-expected-instruction-count 2349
  --validation-fast-subset-size 128

  --validation-fast-interval-steps 375

  --validation-max-new-tokens 512
  --validation-seed 0

  --seed 0
)

nohup env \
  PYTHONNOUSERSITE=1 \
  PYTHONUNBUFFERED=1 \
  PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  TRL_EXPERIMENTAL_SILENCE=1 \
  TOKENIZERS_PARALLELISM=false \
  OMP_NUM_THREADS=1 \
  "$TRAIN_PYTHON" "${TRAIN_ARGS[@]}" \
  > "$CONSOLE_LOG" 2>&1 &

TRAIN_PID=$!

printf '%s\n' "$OUTPUT_DIR" \
  > /tmp/navgpt-main8.1-output-dir

printf '%s\n' "$CONSOLE_LOG" \
  > /tmp/navgpt-main8.1-console-log

printf '%s\n' "$TRAIN_PID" \
  > /tmp/navgpt-main8.1-pid

printf '%s\n' "$TRAIN_MAX_COMPLETION_LENGTH" \
  > /tmp/navgpt-main8.1-max-completion-length

sleep 5

if kill -0 "$TRAIN_PID" 2>/dev/null; then
  echo
  echo "=========================================="
  echo "PASS: 750-step formal training is running"
  echo "=========================================="
  echo "TRAIN_PID=$TRAIN_PID"
  echo "OUTPUT_DIR=$OUTPUT_DIR"
  echo "CONSOLE_LOG=$CONSOLE_LOG"
  echo "MAX_COMPLETION_LENGTH=$TRAIN_MAX_COMPLETION_LENGTH"
  echo "MAX_STEPS=750"
  echo "LEARNING_RATE=2e-6"
else
  echo "FAIL: training exited during startup"
  tail -n 200 "$CONSOLE_LOG"
  exit 1
fi
```
