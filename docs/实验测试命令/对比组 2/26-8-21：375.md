### 跑完之后开始合并
```bash
python - "$EVAL_ROOT" <<'PY'
import json
import sys
from pathlib import Path

root = Path(sys.argv[1])

merged = []

for shard in range(5):
    path = (
        root
        / f"shard-{shard}"
        / "preds"
        / f"submit_R2R_val_unseen_instr_{shard}.json"
    )

    assert path.is_file(), f"missing: {path}"

    rows = json.loads(path.read_text())
    print(f"shard-{shard}: {len(rows)}")
    merged.extend(rows)

ids = [str(x["instr_id"]) for x in merged]

assert len(merged) == 2349, \
    f"prediction count != 2349: {len(merged)}"

assert len(set(ids)) == 2349, \
    f"duplicate instr_id: {len(ids) - len(set(ids))}"

output = root / "merged_predictions.json"

output.write_text(
    json.dumps(
        merged,
        indent=2,
        ensure_ascii=False,
    )
)

print()
print("========================================")
print("PASS MERGE")
print("========================================")
print("predictions =", len(merged))
print("unique ids  =", len(set(ids)))
print("output      =", output)
PY

```

### 测试跑起来
```bash
set -e

PROJECT_ROOT=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/RL-NavGPT
PYTHON=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/venvs/navgpt-rl/bin/python

TRAIN_OUTPUT="$PROJECT_ROOT/outputs/main8.1-ours-qwen14b-fullrl-seed0-2gpu-750steps-lr2e6-20260818-170241"

ADAPTER="$TRAIN_OUTPUT/validation/snapshots/step-375"

VAL_PLAN="$PROJECT_ROOT/datasets/R2R/action_plan_cache/qwen2.5-14b-val-unseen-t0-v1/R2R_val_unseen_action_plans.jsonl"

EVAL_ROOT="$PROJECT_ROOT/datasets/R2R/exprs/ours-qwen14b-grpo-step375-valunseen"

cd "$PROJECT_ROOT/nav_src"

# ============================================================
# 1. 前置检查
# ============================================================

test -f "$ADAPTER/adapter_model.safetensors"
test -f "$ADAPTER/adapter_config.json"
test -f "$ADAPTER/navgpt_adapter_manifest.json"

test -f "$VAL_PLAN"

echo "===== ADAPTER ====="
echo "$ADAPTER"

echo
echo "===== VAL ACTION PLAN ====="
wc -l "$VAL_PLAN"

echo
echo "===== SHARD ANNOTATIONS ====="
for i in 0 1 2 3 4; do
    test -f "$PROJECT_ROOT/datasets/R2R/annotations/R2R_val_unseen_instr_${i}.json"
    echo "PASS shard-$i"
done

mkdir -p "$EVAL_ROOT"

printf '%s\n' "$ADAPTER" > "$EVAL_ROOT/adapter_path.txt"

# ============================================================
# 2. 创建评测 worker
# ============================================================

cat > "$EVAL_ROOT/run_worker.sh" <<'BASH'
#!/usr/bin/env bash

set -e

PROJECT_ROOT=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/RL-NavGPT
PYTHON=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/venvs/navgpt-rl/bin/python

cd "$PROJECT_ROOT/nav_src"

ADAPTER=$(cat "$EVAL_ROOT/adapter_path.txt")

VAL_PLAN="$PROJECT_ROOT/datasets/R2R/action_plan_cache/qwen2.5-14b-val-unseen-t0-v1/R2R_val_unseen_action_plans.jsonl"

for SHARD in "$@"; do

    OUT="$EVAL_ROOT/shard-$SHARD"

    mkdir -p "$OUT"

    echo
    echo "============================================================"
    echo "GPU       = $GPU"
    echo "SHARD     = $SHARD"
    echo "ADAPTER   = $ADAPTER"
    echo "OUTPUT    = $OUT"
    echo "============================================================"

    CUDA_VISIBLE_DEVICES="$GPU" \
    PYTHONNOUSERSITE=1 \
    PYTHONUNBUFFERED=1 \
    TOKENIZERS_PARALLELISM=false \
    "$PYTHON" NavGPT.py \
        --root_dir ../datasets \
        --llm_backend hf \
        --llm_model_name qwen2.5-14b-instruct-1m \
        --local_model_path ../models/Qwen2.5-14B-Instruct-1M \
        --local_adapter_path "$ADAPTER" \
        --local_chat_template auto \
        --local_dtype bf16 \
        --hf_device_map single \
        --navigation_input_mode action_plan \
        --action_plan_cache "$VAL_PLAN" \
        --output_dir "$OUT" \
        --temperature 0 \
        --top_p 0.9 \
        --max_new_tokens 256 \
        --max_iterations 10 \
        --val_env_name "R2R_val_unseen_instr_${SHARD}" \
        --iters -1 \
        --seed 0

    echo
    echo "PASS: shard-$SHARD completed"

done
BASH

chmod +x "$EVAL_ROOT/run_worker.sh"

# ============================================================
# 3. 两张 GPU 并行启动
#
# GPU0: shard 0 -> 2 -> 4
# GPU1: shard 1 -> 3
# ============================================================

setsid -f nohup env \
    GPU=0 \
    EVAL_ROOT="$EVAL_ROOT" \
    bash "$EVAL_ROOT/run_worker.sh" \
    0 2 4 \
    > "$EVAL_ROOT/gpu0.log" \
    2>&1 < /dev/null

setsid -f nohup env \
    GPU=1 \
    EVAL_ROOT="$EVAL_ROOT" \
    bash "$EVAL_ROOT/run_worker.sh" \
    1 3 \
    > "$EVAL_ROOT/gpu1.log" \
    2>&1 < /dev/null

echo
echo "============================================================"
echo "STEP-375 FULL VAL-UNSEEN EVALUATION STARTED"
echo "============================================================"
echo "ADAPTER=$ADAPTER"
echo "EVAL_ROOT=$EVAL_ROOT"
echo
echo "GPU0 log: $EVAL_ROOT/gpu0.log"
echo "GPU1 log: $EVAL_ROOT/gpu1.log"

```
