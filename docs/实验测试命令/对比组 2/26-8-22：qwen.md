### 训练命令

> **历史记录，不是正式 RL evaluator。** 下列命令调用 `NavGPT.py` 的
> LangChain 文本协议，产物为 `official_rl_comparable=false`，不得与 native
> Base/LoRA 指标直接比较。正式四卡命令见
> [`../../训练命令.md` 第 12.1 节](../../训练命令.md#121-唯一正式-native-evaluator)。

```bash
set -e

PROJECT_ROOT=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/RL-NavGPT
PYTHON=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/venvs/navgpt-rl/bin/python

VAL_PLAN="$PROJECT_ROOT/datasets/R2R/action_plan_cache/qwen2.5-14b-val-unseen-t0-v1/R2R_val_unseen_action_plans.jsonl"

EVAL_ROOT="$PROJECT_ROOT/datasets/R2R/exprs/qwen14b-zero-shot-actionplan-valunseen"

cd "$PROJECT_ROOT/nav_src"

test -f "$VAL_PLAN"

mkdir -p "$EVAL_ROOT"

cat > "$EVAL_ROOT/run_worker.sh" <<'BASH'
#!/usr/bin/env bash

set -e

PROJECT_ROOT=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/RL-NavGPT
PYTHON=/UNICOMFS/shskw43_1/.bihu/rl-navgpt/venvs/navgpt-rl/bin/python

cd "$PROJECT_ROOT/nav_src"

VAL_PLAN="$PROJECT_ROOT/datasets/R2R/action_plan_cache/qwen2.5-14b-val-unseen-t0-v1/R2R_val_unseen_action_plans.jsonl"

for SHARD in "$@"; do

    OUT="$EVAL_ROOT/shard-$SHARD"

    mkdir -p "$OUT"

    echo
    echo "============================================================"
    echo "BASE QWEN / NO LORA"
    echo "GPU   = $GPU"
    echo "SHARD = $SHARD"
    echo "============================================================"

    CUDA_VISIBLE_DEVICES="$GPU" \
    PYTHONNOUSERSITE=1 \
    PYTHONUNBUFFERED=1 \
    TOKENIZERS_PARALLELISM=false \
    "$PYTHON" NavGPT.py \
        --root_dir ../datasets \
        --llm_backend hf \
        --llm_model_name qwen2.5-14b-instruct-1m \
        --local_model_path ../models/Qwen2.5-14B-Instruct-1M \
        --local_chat_template auto \
        --local_dtype bf16 \
        --hf_device_map single \
        --navigation_input_mode action_plan \
        --action_plan_cache "$VAL_PLAN" \
        --output_dir "$OUT" \
        --temperature 0 \
        --top_p 0.9 \
        --max_new_tokens 256 \
        --max_iterations 10 \
        --val_env_name "R2R_val_unseen_instr_${SHARD}" \
        --iters -1 \
        --seed 0

    echo "PASS shard-$SHARD"

done
BASH

chmod +x "$EVAL_ROOT/run_worker.sh"

setsid -f nohup env \
    GPU=0 \
    EVAL_ROOT="$EVAL_ROOT" \
    bash "$EVAL_ROOT/run_worker.sh" \
    0 2 4 \
    > "$EVAL_ROOT/gpu0.log" \
    2>&1 < /dev/null

setsid -f nohup env \
    GPU=1 \
    EVAL_ROOT="$EVAL_ROOT" \
    bash "$EVAL_ROOT/run_worker.sh" \
    1 3 \
    > "$EVAL_ROOT/gpu1.log" \
    2>&1 < /dev/null

echo
echo "BASE QWEN ACTION-PLAN EVALUATION STARTED"
echo "EVAL_ROOT=$EVAL_ROOT"
```
